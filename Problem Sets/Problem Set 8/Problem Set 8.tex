\noindent\textbf{1.} (20 pts) (QR factorization.)

\begin{enumerate}[label=\textbf{(\alph{enumi})}, wide=0pt]

\item Apply classical Gram-Schmidt (by hand and calculator) to compute the QR factorization of the following matrices
$$(i)\ \ A = \begin{bmatrix}
4 & 1 \\
3 & 7 
\end{bmatrix},\ \ \ \ (ii)\ \ A = \begin{bmatrix}
1 & 4 & 7 \\
2 & 5 & 2 \\
2 & 2 & -1 
\end{bmatrix}.$$

\color{TangGreen}
\begin{enumerate}[label=(\roman{enumii})]

\item $y_{1} = A_{1} = \begin{bmatrix}
4 \\
3 
\end{bmatrix}$ and $r_{1} = ||y_{1}||_{2} = \sqrt{4^{2} + 3^{2}} = 5$. The first unit vector is
$$q_{1} = \frac{y_{1}}{r_{1}} = \begin{bmatrix}
\frac{4}{5} \\
\frac{3}{5} 
\end{bmatrix}$$
The second unit vector is
$$y_{2} = A_{2} - q_{1}q_{1}^{T}A_{2} = \begin{bmatrix}
1 \\
7 
\end{bmatrix} - \begin{bmatrix}
\frac{4}{5} \\
\frac{3}{5} 
\end{bmatrix}5 = \begin{bmatrix}
-3 \\
4
\end{bmatrix}$$
and
$$q_{2} = \frac{y_{2}}{r_{2}} = \begin{bmatrix}
-\frac{3}{5} \\
\frac{4}{5} 
\end{bmatrix}$$
The QR factorization is
$$A = \begin{bmatrix}
\frac{4}{5} & -\frac{3}{5} \\
\frac{3}{5} & \frac{4}{5} 
\end{bmatrix}\begin{bmatrix}
5 & 5 \\
0 & 5 
\end{bmatrix} = QR$$

\item $y_{1} = A_{1} = \begin{bmatrix}
1 \\
2 \\
2 
\end{bmatrix}$ and $r_{1} = \sqrt{1^{2} + 2^{2} + 2^{2}} = 3$. The first unit vector is
$$q_{1} = \frac{y_{1}}{r_{1}} = \begin{bmatrix}
\frac{1}{3} \\
\frac{2}{3} \\
\frac{2}{3} 
\end{bmatrix}$$
Then, 
$$y_{2} = A_{2} - q_{1}q_{1}^{T}A_{2} = \begin{bmatrix}
4 \\
5 \\
2 
\end{bmatrix} - \begin{bmatrix}
\frac{1}{3} \\
\frac{2}{3} \\
\frac{2}{3} 
\end{bmatrix}6 = \begin{bmatrix}
2 \\
1 \\
-2 
\end{bmatrix}$$
so the unit vector is
$$q_{2} = \frac{y_{2}}{r_{2}} = \begin{bmatrix}
\frac{2}{3} \\
\frac{1}{3} \\
-\frac{2}{3} 
\end{bmatrix}$$
The third vector is
$$y_{3} = A_{3} - q_{2}q_{2}^{T}A_{3} - q_{1}q_{1}^{T}A_{3} = \begin{bmatrix}
7 \\
2 \\
-1 
\end{bmatrix} - \begin{bmatrix}
\frac{2}{3} \\
\frac{1}{3} \\
-\frac{2}{3} 
\end{bmatrix}6 - \begin{bmatrix}
\frac{1}{3} \\
\frac{2}{3} \\
\frac{2}{3} 
\end{bmatrix}3 = \begin{bmatrix}
2 \\
-2 \\
1 
\end{bmatrix}$$
and the third unit vector is
$$q_{3} = \begin{bmatrix}
\frac{2}{3} \\
-\frac{2}{3} \\
\frac{1}{3} 
\end{bmatrix}$$
The QR factorization is
$$A = \begin{bmatrix}
\frac{1}{3} & \frac{2}{3} & \frac{2}{3} \\
\frac{2}{3} & \frac{1}{3} & -\frac{2}{3} \\
\frac{2}{3} & -\frac{2}{3} & \frac{1}{3} 
\end{bmatrix}\begin{bmatrix}
3 & 6 & 3 \\
0 & 3 & 6 \\
0 & 0 & 3 
\end{bmatrix} = QR$$

\end{enumerate}
\color{black}

\item Solve the following least squares problem (by hand and calculator) using QR factorization
$$\begin{bmatrix}
1 & 3 \\
2 & 1 \\
2 & -2 \\
4 & 6 
\end{bmatrix}\begin{bmatrix}
x_{1} \\
x_{2} 
\end{bmatrix} = \begin{bmatrix}
3 \\
1 \\
8 \\
1 
\end{bmatrix}$$
Also compute the 2-norm of the residual. 

\color{TangGreen}
The first vector is $y_{1} = \begin{bmatrix}
1 \\
2 \\
2 \\
4 
\end{bmatrix}$ and $r_{1} = \sqrt{1^{2} + 2^{2} + 2^{2} + 4^{2}} = 5$. The first unit vector is
$$q_{1} = \begin{bmatrix}
\frac{1}{5} \\
\frac{2}{5} \\
\frac{2}{5} \\
\frac{4}{5} 
\end{bmatrix}$$
The second vector is
$$y_{2} = A_{2} - q_{1}q_{1}^{T}A_{2} = \begin{bmatrix}
3 \\
1 \\
-2 \\
6 
\end{bmatrix} - \begin{bmatrix}
\frac{1}{5} \\
\frac{2}{5} \\
\frac{2}{5} \\
\frac{4}{5} 
\end{bmatrix}5 = \begin{bmatrix}
2 \\
-1 \\
-4 \\
2 
\end{bmatrix},\ \ \ \ r_{2} = \sqrt{2^{2} + (-1)^{2} + (-4)^{2} + 2^{2}} = 5$$
and the second unit vector is
$$q_{2} = \begin{bmatrix}
\frac{2}{5} \\
-\frac{1}{5} \\
-\frac{4}{5} \\
\frac{2}{5} 
\end{bmatrix}$$
Given that there isn't a third and forth column in the matrix, those vectors are $A_{3} = \begin{bmatrix}
1 \\
0 \\
0 \\
0 
\end{bmatrix}$ and $A_{4} = \begin{bmatrix}
0 \\
1 \\
0 \\
0 
\end{bmatrix}$. The third vector is
$$y_{3} = A_{3} - q_{2}q_{2}^{T}A_{3} - q_{1}q_{1}^{T}A_{3} = \begin{bmatrix}
1 \\
0 \\
0 \\
0 
\end{bmatrix} - \begin{bmatrix}
\frac{2}{5} \\
-\frac{1}{5} \\
-\frac{4}{5} \\
\frac{2}{5} 
\end{bmatrix}\frac{2}{5} - \begin{bmatrix}
\frac{1}{5} \\
\frac{2}{5} \\
\frac{2}{5} \\
\frac{4}{5} 
\end{bmatrix}\frac{1}{5} = \begin{bmatrix}
\frac{4}{5} \\
0 \\
\frac{6}{25} \\
-\frac{8}{25} 
\end{bmatrix}$$
\color{black}

\end{enumerate}
